server {
    listen 80;
    server_name localhost;
    
    root /usr/share/nginx/html;
    index index.html;

    # --- TỐI ƯU TỐC ĐỘ (PERFORMANCE) ---
    # Nén Gzip để tải trang siêu nhanh
    gzip on;
    gzip_vary on;
    gzip_min_length 10240;
    gzip_proxied expired no-cache no-store private auth;
    gzip_types text/plain text/css text/xml text/javascript application/x-javascript application/xml;
    
    # Cache cực mạnh cho file tĩnh (Ảnh, CSS, JS)
    # Người dùng vào lần 2 sẽ tải tức thì từ bộ nhớ máy họ
    location ~* \.(?:ico|css|js|gif|jpe?g|png|woff2?|eot|ttf|svg)$ {
        expires 6M;
        access_log off;
        add_header Cache-Control "public, max-age=15552000, immutable";
    }

    # --- BẢO MẬT (SECURITY HEADERS) ---
    # Chống click-jacking (không cho web khác nhúng web Tony vào iframe)
    add_header X-Frame-Options "SAMEORIGIN" always;
    # Chống XSS cơ bản
    add_header X-XSS-Protection "1; mode=block" always;
    # Chống đoán định dạng file (MIME-sniffing)
    add_header X-Content-Type-Options "nosniff" always;
    # Ẩn phiên bản Nginx (để hacker không biết version nào mà tấn công)
    server_tokens off;

    # --- ROUTING (REACT APP) ---
    # Fix lỗi 404 khi F5 trang con (SPA Routing)
    location / {
        try_files $uri $uri/ /index.html;
    }
}
